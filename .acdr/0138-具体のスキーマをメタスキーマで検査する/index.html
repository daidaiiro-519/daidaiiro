具体のスキーマをメタスキーマで検査する

決めたこと

基盤が具体のスキーマに求める契約を references/meta-schema.json に置き、基盤のツール check-schemas で、具体のスキーマをすべての深さまで検査する

なぜ決める必要があるか

ボード schema-driven-build の論点6（D）で、基盤はスキーマの中身ではなく書き方の契約を持ち、ツールで検査すると決めた。それまで基盤には注釈ごとの仕様しかなく、具体のスキーマが基盤の契約に従っているかを確かめられなかった。実際に、見本のスキーマには ACDR 0129 で捨てた古い絞りの書き方が19件残り、仕様に無い x-ref のキーは基盤の検査に黙って無視されていた。

どう変わるか

変更前と変更後

何が変更前変更後
基盤が具体のスキーマに求める契約注釈ごとの仕様（annotations ・ view）だけreferences/meta-schema.json。説明と案内（書く単位ごとの description と x-prompt）・ x-generates を持つスキーマの kind ・ 具体の注釈の申告（x-annotations）の形
検査無い基盤のツール check-schemas（引数 dir）。メタスキーマに加えて、注釈の形をすべての深さで、文の型が JMESPath として読めるか、パイプの右が関数の呼び出しか、申告していない注釈、x-ref が指す種類の id を検査する
契約の版──メタスキーマと注釈の仕様は、ビルドのときにツールへ取り込む。同じ版のツールなら、同じスキーマに同じ結果を返す
ポート Querysearch と evaluateparse を追加する（式が JMESPath として読めるかだけを見る）。jmespath の crate はアダプタにだけ置く（ACDR 0122）

比較した案と、採らなかった理由

案採らなかった理由
注釈の形もメタスキーマに書くJSON Schema で全部の深さの注釈を書くと、入れ子の形ごとに定義が必要になり、同じ違反が2回出る。メタスキーマは構造の契約だけを持ち、注釈はツールの処理ですべての深さを見る
メタスキーマを実行のときにファイルから読むツールの版とメタスキーマの版がずれうる。core はファイルを読まない決まりもある
x-generates を持つスキーマだけに x-prompt を求める（見本の schema-meta の契約）部品のスキーマ（共通の $defs）のプロパティも、$ref を通してインスタンスに書かれる。x-prompt を求めないと、そこを書く人と AI に案内が無い

46/// 文の型から {…} の式を取り出す。{{ と }} は文字。
47fn expressions(template: &str) -> Vec<String> {
48 let chars: Vec<char> = template.chars().collect();
49 let (mut out, mut i) = (Vec::new(), 0);
50 while i < chars.len() {
51 let c = chars[i];
52 if (c == '{' || c == '}') && chars.get(i + 1) == Some(&c) {
53 i += 2;
54 continue;
55 }
56 if c != '{' {
57 i += 1;
58 continue;
59 }
60 let (mut depth, mut quote, mut expr) = (1, None::<char>, String::new());
61 i += 1;
62 while i < chars.len() {
63 let d = chars[i];
64 match quote {
65 Some(q) if d == q => quote = None,
66 Some(_) => {}
67 None if "'`\"".contains(d) => quote = Some(d),
68 None if d == '{' => depth += 1,
69 None if d == '}' => {
70 depth -= 1;
71 if depth == 0 {
72 break;
73 }
74 }
75 None => {}
76 }
77 expr.push(d);
78 i += 1;
79 }
80 out.push(expr.trim().to_owned());
81 i += 1;
82 }
83 out
84}

86/// x-view が持つ文の型と式（text ・ cases ・ parts ・ label）。式はそのまま {…} で包んで返す。
87fn templates(xv: &Value) -> Vec<String> {
88 let mut t = Vec::new();
89 let case = |c: &Value, t: &mut Vec<String>| {
90 if let Some(s) = c.get("text").and_then(Value::as_str) {
91 t.push(s.to_owned());

94 t.push(format!("{{{s}}}"));
95 }
96 };

98 t.push(s.to_owned());

102 .and_then(Value::as_array)
103 .into_iter()
104 .flatten()

107 }

116 Value::Array(a) => a.iter().for_each(|c| case(c, &mut t)),

126/// パイプの右が関数の呼び出しでない（|name のような古い絞りの書き方）。
127fn old_filter(expr: &str) -> bool {
128 let mut rest = expr;
129 while let Some(p) = rest.find('|') {
130 let right = rest[p + 1..].trim_start();
131 if let Some(after_or) = right.strip_prefix('|') {
132 // || は「または」
133 rest = after_or;
134 continue;
135 }
136 let id: String = right
137 .chars()
138 .take_while(|c| c.is_ascii_alphanumeric() || *c == '_')
139 .collect();
140 if !id.is_empty() {
141 let after = right[id.len()..].trim_start();
142 if !after.starts_with('(') && !after.starts_with('.') && !after.starts_with('[') {
143 return true;
144 }
145 }
146 rest = &rest[p + 1..];
147 }
148 false
149}

170fn walk(
171 v: &Value,
172 at: &str,
173 name: &str,
174 declared: &BTreeSet<String>,
175 shapes: &Shapes,
176 parse: &dyn Fn(&str) -> Result<(), String>,
177 out: &mut Vec<SchemaFinding>,
178) {
179 let mut push = |at: String, rule: &str, message: String| {
180 out.push(SchemaFinding {
181 schema: name.to_owned(),
182 at,
183 rule: rule.to_owned(),
184 message,
185 })
186 };
187 match v {
188 Value::Object(m) => {
189 let mut children = Vec::new();
190 for (k, c) in m {
191 let here = format!("{at}/{}", pointer_part(k));
192 if !k.starts_with("x-") {
193 if !OPAQUE.contains(&k.as_str()) {
194 children.push((here, c));
195 }
196 continue;
197 }
198 let shape = match k.as_str() {
199 "x-view" => Some(&shapes.view),
200 "x-ref" => Some(&shapes.xref),
201 "x-derive" => Some(&shapes.derive),
202 _ => None,
203 };
204 if let Some(s) = shape {
205 if let Some(e) = first_error(s, c) {
206 push(here.clone(), "注釈の形", e);
207 }
208 }
209 if k == "x-prompt" {
210 let ok = ["read", "write"].iter().all(|f| {
211 c.get(*f)
212 .and_then(Value::as_str)
213 .is_some_and(|s| !s.is_empty())
214 });
215 if !ok {
216 push(here.clone(), "注釈の形", "read と write が無い".into());
217 }
218 }
219 if k == "x-view" && c.is_object() {
220 for t in templates(c) {
221 for e in expressions(&t) {
222 if old_filter(&e) {
223 push(here.clone(), "古い絞りの書き方", format!("{{{e}}} ── パイプの右は関数の呼び出しにする（例：name(@)）"));
224 } else if let Err(err) = parse(&e) {
225 push(
226 here.clone(),
227 "文の型",
228 format!("{{{e}}} ── {}", err.lines().next().unwrap_or("")),
229 );
230 }
231 }
232 }
233 }
234 if !BASE_ANNOTATIONS.contains(&k.as_str()) && !declared.contains(k) {
235 push(
236 here,
237 "申告していない注釈",
238 format!("{k} は基盤の注釈でなく、x-annotations にも無い"),
239 );
240 }
241 }
242 for (here, c) in children {
243 walk(c, &here, name, declared, shapes, parse, out);
244 }
245 }
246 Value::Array(a) => {
247 for (i, c) in a.iter().enumerate() {
248 walk(c, &format!("{at}/{i}"), name, declared, shapes, parse, out);
249 }
250 }
251 _ => {}
252 }
253}
